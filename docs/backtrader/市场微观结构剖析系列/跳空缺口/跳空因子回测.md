```python
        print("计算隔夜跳空因子...")
        
        # 隔夜收益率：今日开盘 / 昨日收盘 - 1
        self.futures_data['overnight_return'] = self.futures_data['open'] / self.futures_data['close'].shift(1) - 1
        
        # 对数隔夜收益率
        self.futures_data['log_overnight_return'] = (
            self.futures_data['log_open'] - self.futures_data['log_close'].shift(1)
        )

            self.futures_data['log_close'] - self.futures_data['log_open']
        )

    def construct_composite_factor(self):
        """构造合成因子（融合昼夜信息）"""
        print("构造昼夜合成因子...")
        
        # 日内黄金分割反转因子（正相关）
        intra_factor = 'INTRA_REVERSAL_20'
        
        # 隔夜跳空因子（负相关，需取负号）
        overnight_factor = 'OVERNIGHT_GAP_10'
        
        # 标准化因子
        self.futures_data[f'{intra_factor}_zscore'] = (
            (self.futures_data[intra_factor] - self.futures_data[intra_factor].rolling(60).mean()) 
            / self.futures_data[intra_factor].rolling(60).std()
        )
        
        self.futures_data[f'{overnight_factor}_zscore'] = (
            (self.futures_data[overnight_factor] - self.futures_data[overnight_factor].rolling(60).mean()) 
            / self.futures_data[overnight_factor].rolling(60).std()
        )
        
        # 合成因子（6:4权重，参考华安证券）
        self.futures_data['COMPOSITE_FACTOR'] = (
            0.6 * self.futures_data[f'{intra_factor}_zscore'] 
            - 0.4 * self.futures_data[f'{overnight_factor}_zscore']  # 取负号
        )
        
        self.factors['COMPOSITE_FACTOR'] = {
            'description': '昼夜合成因子（60%日内反转 + 40%隔夜跳空）',
            'direction': '正'
        }
        
        print("合成因子构造完成")
        return self.futures_data
    
    def backtest_strategy(self, factor_name='COMPOSITE_FACTOR', future_window=5, 
                         transaction_cost=0.0003, initial_capital=1000000):
        """策略回测"""
        print(f"对{factor_name}进行策略回测...")
        
        df = self.futures_data.copy()
        
        # 生成交易信号（每月调仓）
        df['signal'] = 0
        df['position'] = 0
        
        # 每月最后一个交易日调仓
        monthly_dates = df.resample('M').last().index
        monthly_dates = [date for date in monthly_dates if date in df.index]
        
        for i in range(len(monthly_dates) - 1):
            rebalance_date = monthly_dates[i]
            next_rebalance_date = monthly_dates[i + 1]
            
            # 获取调仓日的因子值
            factor_value = df.loc[rebalance_date, factor_name]
            
            if pd.isna(factor_value):
                continue
            
            # 生成信号（简单多头策略：因子值大于0做多，小于0做空）
            signal = 1 if factor_value > 0 else -1
            
            # 在调仓期间保持信号
            mask = (df.index >= rebalance_date) & (df.index < next_rebalance_date)
            df.loc[mask, 'signal'] = signal
            df.loc[mask, 'position'] = signal  # 满仓
        
        # 计算策略收益（考虑交易成本）
        df['strategy_return'] = df['position'].shift(1) * df['close_return']
        
        # 减去年化交易成本（每月调仓一次）
        trade_dates = df[df['position'] != df['position'].shift(1)].index
        for trade_date in trade_dates:
            if trade_date in df.index:
                df.loc[trade_date, 'strategy_return'] -= transaction_cost
        
        # 计算净值
        df['strategy_cum_return'] = (1 + df['strategy_return'].fillna(0)).cumprod()
        df['benchmark_cum_return'] = (1 + df['close_return'].fillna(0)).cumprod()
        
        # 计算绩效指标
        strategy_returns = df['strategy_return'].dropna()
        benchmark_returns = df['close_return'].dropna()
        
        # 年化收益率
        annual_trading_days = 252
        strategy_annual_return = (1 + strategy_returns.mean()) ** annual_trading_days - 1
        benchmark_annual_return = (1 + benchmark_returns.mean()) ** annual_trading_days - 1
        
        # 年化波动率
        strategy_annual_vol = strategy_returns.std() * np.sqrt(annual_trading_days)
        benchmark_annual_vol = benchmark_returns.std() * np.sqrt(annual_trading_days)
        
        # 夏普比率
        risk_free_rate = 0.03  # 3%无风险利率
        strategy_sharpe = (strategy_annual_return - risk_free_rate) / strategy_annual_vol
        benchmark_sharpe = (benchmark_annual_return - risk_free_rate) / benchmark_annual_vol
        
        # 最大回撤
        strategy_cummax = df['strategy_cum_return'].cummax()
        strategy_drawdown = (df['strategy_cum_return'] - strategy_cummax) / strategy_cummax
        strategy_max_drawdown = strategy_drawdown.min()
        
        benchmark_cummax = df['benchmark_cum_return'].cummax()
        benchmark_drawdown = (df['benchmark_cum_return'] - benchmark_cummax) / benchmark_cummax
        benchmark_max_drawdown = benchmark_drawdown.min()
        
        # 胜率
        strategy_win_rate = (strategy_returns > 0).mean()
        
        # 盈亏比
        winning_returns = strategy_returns[strategy_returns > 0]
        losing_returns = strategy_returns[strategy_returns <= 0]
        profit_loss_ratio = abs(winning_returns.mean() / losing_returns.mean()) if len(losing_returns) > 0 else np.inf
        
        performance = {
            '策略年化收益率': strategy_annual_return,
            '基准年化收益率': benchmark_annual_return,
            '策略年化波动率': strategy_annual_vol,
            '基准年化波动率': benchmark_annual_vol,
            '策略夏普比率': strategy_sharpe,
            '基准夏普比率': benchmark_sharpe,
            '策略最大回撤': strategy_max_drawdown,
            '基准最大回撤': benchmark_max_drawdown,
            '策略胜率': strategy_win_rate,
            '策略盈亏比': profit_loss_ratio,
            '净值曲线': df[['strategy_cum_return', 'benchmark_cum_return']]
        }
        
        print(f"回测结果:")
        print(f"  策略年化收益: {strategy_annual_return:.2%}")
        print(f"  基准年化收益: {benchmark_annual_return:.2%}")
        print(f"  策略夏普比率: {strategy_sharpe:.2f}")
        print(f"  策略最大回撤: {strategy_max_drawdown:.2%}")
        print(f"  策略胜率: {strategy_win_rate:.2%}")
        print(f"  策略盈亏比: {profit_loss_ratio:.2f}")
        
        return performance

    def calculate_liquidity_adjusted_factors(self):

        print("计算流动性调整因子...")
        
        # 计算流动性指标
        self.futures_data['volume_ma'] = self.futures_data['volume'].rolling(20).mean()
        self.futures_data['volume_ratio'] = self.futures_data['volume'] / self.futures_data['volume_ma']
        self.futures_data['turnover_ma'] = self.futures_data['turnover'].rolling(20).mean()
        self.futures_data['turnover_ratio'] = self.futures_data['turnover'] / self.futures_data['turnover_ma']
        
        # 流动性加权跳空因子
        for window in [5, 10, 20]:
            # 基础跳空因子
            gap_factor = f'JUMP_{window}'
            liquidity_factor = f'liquidity_adj_JUMP_{window}'
            
            # 使用成交量倒数加权（流动性越差，权重越低）
            weights = 1 / self.futures_data['volume_ratio'].clip(lower=0.5, upper=2)
            weighted_gap = self.futures_data['log_gap_abs'].rolling(window).apply(
                lambda x: np.average(x, weights=weights.loc[x.index[-len(x):]]), raw=False
            )
            
            self.futures_data[liquidity_factor] = -weighted_gap
            self.factors[liquidity_factor] = {
                'window': window,
                'description': f'流动性调整的{window}日跳空因子',
                'direction': '正'
            }
        
        print("流动性调整因子计算完成")
        return self.futures_data
    
    def calculate_volatility_adjusted_factors(self):
        """考虑波动率的调整因子"""
        print("计算波动率调整因子...")
        
        # 计算波动率
        self.futures_data['volatility_20d'] = self.futures_data['close_return'].rolling(20).std() * np.sqrt(252)
        
        for window in [5, 10, 20]:
            # 基础跳空因子
            gap_factor = f'JUMP_{window}'
            vol_factor = f'vol_adj_JUMP_{window}'
            
            # 跳空幅度相对于波动率的比率
            avg_gap = self.futures_data['log_gap_abs'].rolling(window).mean()
            avg_vol = self.futures_data['volatility_20d'].rolling(window).mean()
            
            self.futures_data[vol_factor] = -avg_gap / (avg_vol + 1e-6)
            self.factors[vol_factor] = {
                'window': window,
                'description': f'波动率调整的{window}日跳空因子',
                'direction': '正'
            }
        
        print("波动率调整因子计算完成")
        return self.futures_data
    
    def calculate_implied_funding_rate_effect(self):
        """考虑资金成本影响"""
        print("计算资金成本调整...")
        
        # 模拟资金成本（期货贴水/升水）
        # 实际应用中应从期货实际数据计算
        self.futures_data['basis'] = (self.futures_data['close'] - self.futures_data['close'].rolling(20).mean()) / self.futures_data['close'].rolling(20).mean()

            funding_factor = f'funding_adj_JUMP_{window}'
            
            # 当期货大幅贴水时，跳空效应可能减弱
            basis_effect = np.where(self.futures_data['basis'] < -0.01, 0.5, 1.0)
            basis_effect = np.where(self.futures_data['basis'] > 0.01, 1.5, basis_effect)
            
            self.futures_data[funding_factor] = self.futures_data[gap_factor] * basis_effect

                'description': f'资金成本调整的{window}日跳空因子',
                'direction': '正'
            }

        return self.futures_data
    
    def construct_multi_timeframe_factors(self):

        # 不同时间周期的跳空因子

        for tf_name, (start_time, end_time) in timeframes.items():

                self.futures_data['multi_timeframe_factor'] = intra_vol_z - overnight_gap_z
```

```python
4. 流动性风险
   - 小盘股期货流动性较差
   - 大单冲击成本较高

5. 保证金管理
   - 跳空可能触发强平
   - 需动态调整保证金
"""

# 实际期货交易接口示例
class IMActualTrading:
    """中证1000股指期货实际交易类"""
    
    def __init__(self):
        self.contract_multiplier = 200  # 合约乘数
        self.margin_rate = 0.12  # 保证金率
        self.transaction_fee = 0.000023  # 手续费率
    
    def calculate_position_size(self, capital, volatility):
        """基于波动率的仓位计算"""
        # 凯利公式变体
        risk_per_trade = 0.02  # 单笔风险2%
        position_value = capital * risk_per_trade / volatility
        contracts = int(position_value / (self.contract_multiplier * self.margin_rate))
        return max(1, contracts)
    
    def backtest_with_leverage(self, signals, prices, initial_capital=1000000):
        """带杠杆的回测"""
        positions = signals.shift(1)  # 滞后一期
        returns = positions * prices.pct_change()
        
        # 计算保证金占用
        margin_required = abs(positions) * prices * self.contract_multiplier * self.margin_rate
        
        # 计算杠杆后收益
        levered_returns = returns * (initial_capital / margin_required).clip(upper=3)  # 最大3倍杠杆
        
        # 扣除手续费
        trade_changes = (positions != positions.shift(1)).astype(int)
        levered_returns -= trade_changes * self.transaction_fee
        
        return levered_returns.cumsum()
```

```python
class RiskManagement:

        """动态仓位调整"""
        if current_drawdown < self.max_drawdown_limit:
            return 0  # 暂停交易
        
        # 波动率调整仓位
        adj_position = self.position_limit * (0.02 / volatility)  # 目标波动率2%
        return min(adj_position, self.position_limit)
    
    def gap_size_filter(self, gap_ratio, threshold=0.03):
        """跳空大小过滤"""
        # 过大跳空可能不适合交易

        return True
    
    def liquidity_filter(self, volume, avg_volume):
        """流动性过滤"""
        if volume < avg_volume * 0.5:
```
